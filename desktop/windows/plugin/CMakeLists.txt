cmake_minimum_required(VERSION 3.20)
project(nvu_scs_telemetry LANGUAGES CXX)
set(NVU_PLUGIN_VERSION "" CACHE STRING "Plugin version stamped into the DLL (build-plugin.cjs passes package.json nvu.pluginVersion)")
if(NOT NVU_PLUGIN_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Pass -DNVU_PLUGIN_VERSION=x.y.z (use: npm --prefix desktop/windows run plugin:build)")
endif()
add_library(nvu_scs_telemetry SHARED nvu_scs_telemetry.cpp)
target_compile_definitions(nvu_scs_telemetry PRIVATE NVU_PLUGIN_VERSION="${NVU_PLUGIN_VERSION}")
target_compile_features(nvu_scs_telemetry PRIVATE cxx_std_17)
target_include_directories(nvu_scs_telemetry PRIVATE vendor/include)
target_link_libraries(nvu_scs_telemetry PRIVATE shell32 ole32 uuid)
set_target_properties(nvu_scs_telemetry PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(MSVC)
  # Exports come from the .def file (see the note above NVU_SDK_ENTRY in nvu_scs_telemetry.cpp).
  target_sources(nvu_scs_telemetry PRIVATE nvu_scs_telemetry.def)
  set_property(TARGET nvu_scs_telemetry PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(nvu_scs_telemetry PRIVATE -static -static-libgcc -static-libstdc++)
endif()
